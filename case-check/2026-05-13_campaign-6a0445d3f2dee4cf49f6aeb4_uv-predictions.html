Query A — IAP & AdRev

SELECT
  submit_date,
  body.dep_version,
  MAX(body.target_game_id)      AS target_game_id,
  COUNT(*)                      AS start_count,
  AVG(body.dep_p)               AS avg_dep_pred,
  AVG(body.dep_adj)             AS avg_dep_adj,
  AVG(body.dep_value)           AS avg_dep_value,
  AVG(body.adrev_p)             AS avg_adrev_pred,
  AVG(body.adrev_adj)           AS avg_adrev_adj,
  AVG(body.adrev_value)         AS avg_adrev_value,
  AVG(body.cst)                 AS avg_cost,
  AVG(body.max_cost)            AS avg_max_cost,
  AVG(body.actual_max_cost)     AS avg_actual_max_cst
FROM `unity-ai-data-prd.mz_dcpi_raw.mz_dcpi_prediction_v1`
WHERE submit_date BETWEEN "2026-05-13" AND "2026-05-30"
  AND body.ar_ts >= "2026-05-13 00:00:00"
  AND body.dep_p > 0
  AND body.campaign_id = "6a0445d3f2dee4cf49f6aeb4"
GROUP BY submit_date, body.dep_version
ORDER BY submit_date

Query B — CPE (level_complete / purchase / retention)

SELECT
  submit_date,
  body.app_event_type,
  body.app_event_model_version  AS model_version,
  MAX(body.target_game_id)      AS target_game_id,
  COUNT(*)                      AS start_count,
  AVG(body.app_event_p)         AS avg_cpe_pred,
  AVG(body.app_event_adj)       AS avg_cpe_adj
FROM `unity-ai-data-prd.mz_dcpi_raw.mz_dcpi_prediction_v1`
WHERE submit_date BETWEEN "2026-05-13" AND "2026-05-30"
  AND body.ar_ts >= "2026-05-13 00:00:00"
  AND body.app_event_type IN ("level_complete", "purchase", "retention")
  AND body.app_event_p > 0
  AND body.campaign_id = "6a0445d3f2dee4cf49f6aeb4"
GROUP BY submit_date, body.app_event_type, body.app_event_model_version
ORDER BY submit_date
